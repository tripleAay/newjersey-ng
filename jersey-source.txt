function roundMoney(value: number) {
  return Math.round(value * 100) / 100;
}

function isMoney(value: unknown): value is number {
  return (
    typeof value === "number" &&
    Number.isFinite(value) &&
    value >= 0 &&
    value <= Number.MAX_SAFE_INTEGER / 100
  );
}

function createCartItemId(
  productId: string,
  selections: SelectedProductOption[],
  artworkType: ArtworkType
) {
  const signature = selections
    .map(
      (selection) =>
        `${encodeURIComponent(selection.optionId)}:${encodeURIComponent(
          selection.choiceId
        )}`
    )
    .sort()
    .join("|");

  return `${encodeURIComponent(productId)}__${signature}__${artworkType}`;
}

export default function ProductConfigurator(
  props: ProductConfiguratorProps
) {
  return (
    <ConfiguratorContent
      key={`${props.product.id}:${props.product.designFee ?? "unset"}`}
      {...props}
    />
  );
}

function ConfiguratorContent({
  product,
  quoteRequestHref,

  const { addItem, hydrated } = useCart();

  const [selections, setSelections] = useState<
    Record<string, string>
  >(() => {
    const initial: Record<string, string> = {};

    for (const option of product.options ?? []) {
      if (option.required && option.choices[0]) {
        initial[option.id] = option.choices[0].id;
      }
    }

    return initial;
  });

  const [artworkType, setArtworkType] =
    useState<ArtworkType>("customer-supplied");

  const [added, setAdded] = useState(false);
  const [error, setError] = useState("");

  const selectedOptions = useMemo<
    SelectedProductOption[]
  >(() => {
    const result: SelectedProductOption[] = [];

    for (const option of product.options ?? []) {
      const choice = option.choices.find(
        (candidate) =>
          candidate.id === selections[option.id]
      );

      if (!choice) continue;

      result.push({
        optionId: option.id,
        optionLabel: option.label,
        choiceId: choice.id,
        choiceLabel: choice.label,
        priceModifier: choice.priceModifier ?? 0,
      });
    }

    return result;
  }, [product.options, selections]);

  const basePrice = product.pricing.basePrice ?? 0;

  const optionsTotal = roundMoney(
    selectedOptions.reduce(
      (sum, option) => sum + option.priceModifier,
      0
    )
  );

  // Production cost for one configured job.
  const unitPrice = roundMoney(basePrice + optionsTotal);

  const designAvailable = isMoney(product.designFee);

  const designFee =
    artworkType === "design-needed" && designAvailable
      ? roundMoney(product.designFee!)
      : 0;

  const estimatedTotal = roundMoney(
    unitPrice + designFee
  );

  const pricingValid =
    isMoney(product.pricing.basePrice) &&
    selectedOptions.every((option) =>
      isMoney(option.priceModifier)
    ) &&
    isMoney(optionsTotal) &&
    isMoney(unitPrice) &&
    isMoney(estimatedTotal);

  const artworkValid =
    artworkType === "customer-supplied" ||
    designAvailable;

  function resetFeedback() {
    setAdded(false);
    setError("");
  }

  function chooseOption(
    optionId: string,
    choiceId?: string
  ) {
    setSelections((current) => {
      const next = { ...current };

      if (choiceId === undefined) {
        delete next[optionId];
      } else {
        next[optionId] = choiceId;
      }

      return next;
    });

    resetFeedback();
  }

  function chooseArtwork(type: ArtworkType) {
    setArtworkType(type);
    resetFeedback();
  }

  function handleAddToOrder() {
    resetFeedback();

    if (!hydrated) {
      setError("Your order is still loading.");
      return;
    }

    if (
      product.published === false ||
      product.pricing.type === "quote"
    ) {
      setError("This product is unavailable for purchase.");
      return;
    }

    const missing = (product.options ?? []).find(
      (option) =>
        option.required &&
        !option.choices.some(
          (choice) =>
            choice.id === selections[option.id]
        )
    );

    if (missing) {
      setError(`Choose an option for ${missing.label}.`);
      return;
    }

    if (!pricingValid || !artworkValid) {
      setError(
        "Pricing is unavailable for these selections."
      );
      return;
    }

    try {
      addItem({
        cartItemId: createCartItemId(
          product.id,
          selectedOptions,
          artworkType
        ),
        productId: product.id,
        slug: product.slug,
        name: product.name,
        image: product.image || product.images?.[0] || "",
        quantity: 1,
        basePrice: roundMoney(basePrice),
        optionsTotal,

        // Keep production totals separate from artwork.
        unitPrice,
        totalPrice: unitPrice,

        selections: selectedOptions,
        artwork: {
          type: artworkType,
          designFee,
        },
      });

      setAdded(true);
    } catch {
      setError(
        "Unable to add this item. Please try again."
      );
    }
  }

  if (product.pricing.type === "quote") {
    return (
      <div className="rounded-2xl border border-black/10 bg-white p-5 sm:p-7">
        <h2 className="text-xl font-semibold tracking-tight">
          Custom production
        </h2>

        <p className="mt-3 text-sm leading-7 text-[#625D55]">
          This product requires pricing based on your
          specifications.
        </p>

        {quoteRequestHref ? (
          <Link
            href={quoteRequestHref}
            className={`mt-6 inline-flex min-h-12 w-full items-center justify-center rounded-xl bg-[#F58220] px-5 text-sm font-semibold text-[#181818] transition-colors hover:bg-[#D96B0D] ${focus}`}
          >

        ) : (
          <p className="mt-4 text-sm text-[#625D55]">

        )}

  );

class RequestError extends Error {
  constructor(
    message: string,
    public status = 400
  ) {
    super(message);
    this.name = "RequestError";
  }
}

function isRecord(
  value: unknown
): value is Record<string, unknown> {
  return (
    typeof value === "object" &&
    value !== null &&
    !Array.isArray(value)
  );
}

function requireRecord(
  value: unknown,
  message: string
): Record<string, unknown> {
  if (!isRecord(value)) {
    throw new RequestError(message);
  }

  return value;
}

function readText(
  value: unknown,
  label: string,
  maxLength: number,
  required = false
): string | null {
  if (value === undefined || value === null) {
    if (required) {
      throw new RequestError(`${label} is required.`);
    }

    return null;
  }

  if (typeof value !== "string") {
    throw new RequestError(`${label} must be text.`);
  }

  const text = value.trim();

  if (!text) {
    if (required) {
      throw new RequestError(`${label} is required.`);
    }

    return null;
  }

  if (text.length > maxLength) {
    throw new RequestError(`${label} is too long.`);
  }

  return text;
}

function requiredText(
  value: unknown,
  label: string,
  maxLength: number
): string {
  return readText(value, label, maxLength, true)!;
}

function isMoney(value: unknown): value is number {
  return (
    typeof value === "number" &&
    Number.isFinite(value) &&
    value >= 0 &&
    value <= MAX_MONEY
  );
}

function roundMoney(value: number) {
  return Math.round(value * 100) / 100;
}

function sameMoney(left: number, right: number) {
  return (
    Math.round(left * 100) ===
    Math.round(right * 100)
  );
}

function parseDatabaseMoney(value: unknown): number {
  if (
    (
      typeof value !== "number" &&
      typeof value !== "string"
    ) ||
    (typeof value === "string" && !value.trim())
  ) {
    throw new Error("Invalid database amount.");
  }

  const amount = Number(value);

  if (!isMoney(amount)) {
    throw new Error("Invalid database amount.");
  }

  return roundMoney(amount);
}

function readDeadline(value: unknown): string | null {
  const deadline = readText(
    value,
    "Preferred deadline",
    10
  );

  if (!deadline) return null;

  if (!/^\d{4}-\d{2}-\d{2}$/.test(deadline)) {
    throw new RequestError("Enter a valid deadline.");
  }

  const parsed = new Date(`${deadline}T00:00:00.000Z`);

  if (
    !Number.isFinite(parsed.getTime()) ||
    parsed.toISOString().slice(0, 10) !== deadline ||
    deadline < "0001-01-01"
  ) {
    throw new RequestError("Enter a valid deadline.");
  }

  return deadline;
}

function generateOrderNumber() {
  const date = new Date()
    .toISOString()
    .slice(2, 10)
    .replaceAll("-", "");

  const reference = randomUUID()
    .replaceAll("-", "")
    .slice(0, 16)
    .toUpperCase();

  return `NJ-${date}-${reference}`;
}

function createCartItemId(
  productId: string,
  selections: SelectedProductOption[],
  artworkType: string
) {
  const signature = selections
    .map(
      (selection) =>
        `${encodeURIComponent(selection.optionId)}:${encodeURIComponent(
          selection.choiceId
        )}`
    )
    .sort()
    .join("|");

  return `${encodeURIComponent(productId)}__${signature}__${artworkType}`;
}

function rebuildItem(value: unknown) {
  const item = requireRecord(
    value,
    "One or more order items are invalid."
  );

  const productId = requiredText(
    item.productId,
    "Product ID",
    200
  );

  const product = getProductById(productId);

  if (!product || product.published === false) {
    throw new RequestError(
      "An item in your order is no longer available."
    );
  }

  if (product.pricing.type === "quote") {
    throw new RequestError(
      `${product.name} requires a separate quote request.`
    );
  }

  const quantity = item.quantity;

  if (
    typeof quantity !== "number" ||
    !Number.isSafeInteger(quantity) ||
    quantity < 1 ||
    quantity > MAX_JOBS_PER_ITEM
  ) {
    throw new RequestError(
      `Choose between 1 and ${MAX_JOBS_PER_ITEM} jobs for ${product.name}.`
    );
  }

  if (!Array.isArray(item.selections)) {
    throw new RequestError(
      `Invalid selections for ${product.name}.`
    );
  }

  const productOptions = product.options ?? [];

  if (item.selections.length > productOptions.length) {
    throw new RequestError(
      `Too many selections for ${product.name}.`
    );
  }

  const requestedChoices = new Map<string, string>();

  for (const rawSelection of item.selections) {
    const selection = requireRecord(
      rawSelection,
      `Invalid selections for ${product.name}.`
    );

    const optionId = requiredText(
      selection.optionId,
      "Option ID",
      200
    );

    const choiceId = requiredText(
      selection.choiceId,
      "Choice ID",
      200
    );

    if (requestedChoices.has(optionId)) {
      throw new RequestError(
        `Duplicate option selected for ${product.name}.`
      );
    }

    const option = productOptions.find(
      (candidate) => candidate.id === optionId
    );

    if (
      !option ||
      !option.choices.some(
        (choice) => choice.id === choiceId
      )
    ) {
      throw new RequestError(
        `A selected option is unavailable for ${product.name}.`
      );
    }

    requestedChoices.set(optionId, choiceId);
  }

  const selections: SelectedProductOption[] = [];

  for (const option of productOptions) {
    const choiceId = requestedChoices.get(option.id);

    if (!choiceId) {
      if (option.required) {
        throw new RequestError(
          `Select ${option.label} for ${product.name}.`
        );
      }

      continue;
    }

    const choice = option.choices.find(
      (candidate) => candidate.id === choiceId
    );

    if (!choice) {
      throw new RequestError(
        `A selected option is unavailable for ${product.name}.`
      );
    }

    const modifier = choice.priceModifier ?? 0;

    if (!isMoney(modifier)) {
      throw new Error(
        `Invalid catalogue modifier: ${product.id}/${option.id}`
      );
    }

    selections.push({
      optionId: option.id,
      optionLabel: option.label,
      choiceId: choice.id,
      choiceLabel: choice.label,
      priceModifier: roundMoney(modifier),
    });
  }

  const artwork = requireRecord(
    item.artwork,
    `Select an artwork option for ${product.name}.`
  );

  const artworkType = artwork.type;

  if (
    artworkType !== "customer-supplied" &&
    artworkType !== "design-needed"
  ) {
    throw new RequestError(
      `Select a valid artwork option for ${product.name}.`
    );
  }

  if (!isMoney(product.pricing.basePrice)) {
    throw new Error(
      `Invalid catalogue base price: ${product.id}`
    );
  }

  const basePrice = roundMoney(
    product.pricing.basePrice
  );

  const optionsTotal = roundMoney(
    selections.reduce(
      (sum, selection) =>
        sum + selection.priceModifier,
      0
    )
  );

  const unitPrice = roundMoney(
    basePrice + optionsTotal
  );

  const totalPrice = roundMoney(
    unitPrice * quantity
  );

  if (
    !isMoney(optionsTotal) ||
    !isMoney(unitPrice) ||
    !isMoney(totalPrice)
  ) {
    throw new RequestError(
      "The order amount is too large."
    );
  }

  return {
    cart_item_id: createCartItemId(
      product.id,
      selections,
      artworkType
    ),
    product_id: product.id,
    product_slug: product.slug,
    product_name: product.name,
    product_image:
      product.image || product.images?.[0] || null,
    quantity,
    base_price: basePrice,
    options_total: optionsTotal,
    unit_price: unitPrice,
    total_price: totalPrice,
    selections,
    artwork_type: artworkType,
    design_fee: 0,
  };
}

        {

            "Unable to confirm the saved order. Check whether it was created before submitting again.",

      );

      {

          "Unable to create the production request.",
      },
      { status: 500 }
    );
  }
}

function formatMoney(
  value: number | string | null,
  currency = "NGN"
) {
  if (
    value === null ||
    (typeof value === "string" && value.trim() === "")
  ) {
    return "Not set";
  }

  const amount = Number(value);

  if (!Number.isFinite(amount)) return "Not set";

  try {
    return new Intl.NumberFormat("en-NG", {
      style: "currency",
      currency,
      maximumFractionDigits: 2,
    }).format(amount);
  } catch {
    return `${currency} ${amount.toLocaleString("en-NG")}`;
  }
}

function formatPrice(product: {

function getProductImages(
  mainImage: unknown,
  galleryImages: unknown
): string[] {
  const candidates = [
    mainImage,
    ...(Array.isArray(galleryImages) ? galleryImages : []),
  ];

  const images: string[] = [];

  for (const candidate of candidates) {
    if (typeof candidate !== "string") continue;

    const value = candidate.trim();
    if (!value) continue;

    try {
      const url = new URL(value);

      if (
        url.protocol === "https:" &&
        !url.username &&
        !url.password
      ) {
        images.push(value);
      }
    } catch {
      // Ignore invalid image URLs.
    }
  }

  return [...new Set(images)];
}

"use server";

import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";

import { createClient } from "@/app/lib/newjersey/supabase/server";
import { getSupabaseAdmin } from "@/app/lib/newjersey/supabase/admin";

const UUID =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export async function startBankTransfer(
  formData: FormData
): Promise<void> {
  const orderId = formData.get("orderId");

  if (
    typeof orderId !== "string" ||
    !UUID.test(orderId)
  ) {
    redirect("/shop/order");
  }

  const path = `/shop/order/${orderId}/payment`;

  const auth = await createClient();

  const {
    data: { user },
    error: authError,
  } = await auth.auth.getUser();

  if (authError || !user) {
    redirect(`/login?next=${encodeURIComponent(path)}`);
  }

  const supabase = getSupabaseAdmin();

  const { error } = await supabase.rpc(
    "start_bank_transfer",
    {
      p_order_id: orderId,
      p_customer_id: user.id,
    }
  );

  if (error) {
    console.error("[Start bank transfer]", error);

    const knownMessages = [
      "Order not found.",
      "This order is not available for payment.",
      "Bank transfer is currently unavailable.",
    ];

    const message = knownMessages.includes(error.message)
      ? error.message
      : "Payment instructions could not be created. Refresh the page before trying again.";

    redirect(
      `${path}?error=${encodeURIComponent(message)}`
    );
  }

  revalidatePath(path);

  redirect(path);
}
